function doGet(e) {

    sheetSiswa = ss.insertSheet("Master Siswa");

  }
  var dataSiswaArray = sheetSiswa.getDataRange().getDisplayValues();
  var siswaResult = [];
  for (var i = 1; i < dataSiswaArray.length; i++) {
    if (dataSiswaArray[i][0] !== "") {
      siswaResult.push({ nis: dataSiswaArray[i][0], nama: dataSiswaArray[i][1], kelas: dataSiswaArray[i][2], no_wa: dataSiswaArray[i][3] || "" });

  }

  var absenResult = {};

    var tgl = dataLogArray[j][0];

  }

  var dataAnekdotArray = sheetAnekdot.getDataRange().getDisplayValues();
  var anekdotResult = [];
  for (var m = 1; m < dataAnekdotArray.length; m++) {
    if (dataAnekdotArray[m][0] !== "") {
      anekdotResult.push({
        id: dataAnekdotArray[m][0], tanggal: dataAnekdotArray[m][1], nis: dataAnekdotArray[m][2],
        nama: dataAnekdotArray[m][3], kelas: dataAnekdotArray[m][4], kategori: dataAnekdotArray[m][5],

}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var aksi = e.parameter.aksi; 
  var pinUser = e.parameter.pin || "";
  
  if ((aksi === "edit_siswa" || aksi === "hapus_siswa" || aksi === "hapus_semua" || aksi === "hapus_anekdot") && pinUser !== SERVER_PIN) {
    return ContentService.createTextOutput(JSON.stringify({success: false, message: "Akses Ditolak! PIN Salah."})).setMimeType(ContentService.MimeType.JSON);

      if (!sheetMaster) { sheetMaster = ss.insertSheet("Master Siswa"); sheetMaster.appendRow(["NIS", "Nama Lengkap", "Kelas", "No. WA Ortu"]); }

      return createJsonResponse(true, "Siswa Baru Tersimpan");
    }
    
    else if (aksi === "edit_siswa") {
      var sheetMaster = ss.getSheetByName("Master Siswa");
      var data = sheetMaster.getDataRange().getDisplayValues();
      for (var i = 1; i < data.length; i++) {
        if (data[i][0] === e.parameter.nis_lama) {
          sheetMaster.getRange(i+1, 1).setValue(e.parameter.nis);
          sheetMaster.getRange(i+1, 2).setValue(e.parameter.nama);
          sheetMaster.getRange(i+1, 3).setValue(e.parameter.kelas);
          sheetMaster.getRange(i+1, 4).setValue("'" + (e.parameter.no_wa || ""));
          break;
        }
      }
      return createJsonResponse(true, "Siswa Berhasil Diedit");
    }

    else if (aksi === "hapus_siswa") {
      var sheetMaster = ss.getSheetByName("Master Siswa");
      var data = sheetMaster.getDataRange().getDisplayValues();
      for (var i = 1; i < data.length; i++) {
        if (data[i][0] === e.parameter.nis) {
          sheetMaster.deleteRow(i+1);
          break;
        }
      }
      return createJsonResponse(true, "Siswa Berhasil Dihapus");
    }

    else if (aksi === "hapus_semua") {
      var sheetMaster = ss.getSheetByName("Master Siswa");
      var lastRow = sheetMaster.getLastRow();
      if (lastRow > 1) sheetMaster.deleteRows(2, lastRow - 1);

      var statusRaw = e.parameter.status; var waktu = e.parameter.waktu;

        if (logData[k][0] === tanggalStr && logData[k][2] === nis) sheetLog.deleteRow(k + 1);
      }
      sheetLog.appendRow(["'" + tanggalStr, waktu, nis, nama, kelas, statusRaw]);

      var parts = tanggalStr.split('-');
      var tahun = parseInt(parts[0]); var bulanInt = parseInt(parts[1]); var hari = parseInt(parts[2]);
      var namaBulan = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];
      var sheetName = "Rekap " + namaBulan[bulanInt - 1] + " " + tahun;
      var sheet = ss.getSheetByName(sheetName);
      var daysInMonth = new Date(tahun, bulanInt, 0).getDate(); 
      
      if (!sheet) {
        sheet = ss.insertSheet(sheetName);

        for (var i = 1; i <= daysInMonth; i++) headers.push(i);

      var mark = "H"; var color = "#d1fae5";

        sheet.getRange(rowIndex, colEnd + 3).setFormula('=COUNTIF(' + rangeStr + ', "I")').setBackground("#fef3c7").setFontWeight("bold");

          if (dataAnek[n][0] === e.parameter.id) { sheetAnekdot.deleteRow(n + 1); break; }
        }

      var sheetKategori = ss.getSheetByName("Master Kategori");
      if (!sheetKategori) { sheetKategori = ss.insertSheet("Master Kategori"); sheetKategori.appendRow(["Kategori"]); }

      var dataKategori = sheetKategori.getDataRange().getDisplayValues();
      for (var p = 1; p < dataKategori.length; p++) {
        if (dataKategori[p][0].toLowerCase() === kategoriBaru.toLowerCase()) {
          return createJsonResponse(false, "Kategori tersebut sudah ada.");
        }
      }
      var kategoriBawaan = ["Sosial-Emosional", "Akademik", "Kedisiplinan", "Prestasi", "Kesehatan"];
      if (kategoriBawaan.some(function(k) { return k.toLowerCase() === kategoriBaru.toLowerCase(); })) {

      sheetKategori.appendRow([kategoriBaru]);

      return createJsonResponse(true, "Jenis Kesalahan Baru Tersimpan");

  } catch (error) {
    return createJsonResponse(false, error.toString());
  }
}

// Fungsi Bantuan Pembuat JSON
function createJsonResponse(isSuccess, msg) {
  var data = { success: isSuccess, message: msg };
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function ambilKategoriCustom() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetKategori = ss.getSheetByName("Master Kategori");
  if (!sheetKategori) {
    sheetKategori = ss.insertSheet("Master Kategori");
    sheetKategori.appendRow(["Kategori"]);
  }
  var data = sheetKategori.getDataRange().getDisplayValues();
  var hasil = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] !== "") hasil.push(data[i][0]);
  }
  return hasil;
}

function ambilJenisCustom() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetJenis = ss.getSheetByName("Master Jenis");
  if (!sheetJenis) {
    sheetJenis = ss.insertSheet("Master Jenis");
    sheetJenis.appendRow(["Kategori", "Jenis"]);
  }
  var data = sheetJenis.getDataRange().getDisplayValues();
  var hasil = {};
  for (var i = 1; i < data.length; i++) {
    var kategori = data[i][0]; var jenis = data[i][1];
    if (!kategori || !jenis) continue;
    if (!hasil[kategori]) hasil[kategori] = [];
    hasil[kategori].push(jenis);
  }
  return hasil;
}

function kirimNotifikasiWA(nis, nama, status, waktu, tanggal) {
  try {
    if (!FONNTE_TOKEN || FONNTE_TOKEN.indexOf("TEMPEL_TOKEN") !== -1) return;

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetMaster = ss.getSheetByName("Master Siswa");
    if (!sheetMaster) return;

    var data = sheetMaster.getDataRange().getDisplayValues();
    var noWa = "";
    var ditemukan = false;
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] === nis) {
        ditemukan = true;
        noWa = data[i][3] || "";
        break;
      }
    }
    if (!ditemukan || !noWa) return;

    var nomorTujuan = noWa.trim();
    if (nomorTujuan.charAt(0) === "0") nomorTujuan = "62" + nomorTujuan.substring(1);

    var pesan = "";
    if (status === "Terlambat") {
      pesan = "Assalamu'alaikum, Bapak/Ibu Wali Murid.\nAnanda *" + nama + "* tercatat *TERLAMBAT* hadir di sekolah pada " + tanggal + " pukul " + waktu + ".\nMohon menjadi perhatian bersama. Terima kasih.";
    } else if (status === "Alpa") {
      pesan = "Assalamu'alaikum, Bapak/Ibu Wali Murid.\nAnanda *" + nama + "* tercatat *TIDAK HADIR (Alpa)* di sekolah pada " + tanggal + ".\nMohon konfirmasi ke wali kelas apabila ada keterangan (sakit/izin). Terima kasih.";
    }
    if (!pesan) return;

    UrlFetchApp.fetch("https://api.fonnte.com/send", {
      method: "post",
      headers: { "Authorization": FONNTE_TOKEN },
      payload: { "target": nomorTujuan, "message": pesan },
      muteHttpExceptions: true
    });

  } catch (err) {
    Logger.log("ERROR TAK TERDUGA saat kirim WA: " + err.toString());
  }
}

// ===== BARU: HITUNG & KIRIM PENGINGAT KE GURU JIKA KATEGORI ANEKDOT YANG SAMA SUDAH 3x =====
function cekDanIngatkanGuruAnekdot(nis, nama, kelas, kategori, jenis_terbaru) {
  try {
    if (!kategori) return;

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetAnekdot = ss.getSheetByName("Anekdot");
    if (!sheetAnekdot) return;

    // 1. KUNCI PERBAIKAN: Paksa Google menyimpan baris baru sebelum membacanya
    SpreadsheetApp.flush();

    var data = sheetAnekdot.getDataRange().getDisplayValues();
    var jumlah = 0;
    var daftarKejadian = []; 
    
    for (var i = 1; i < data.length; i++) {
      // Kolom: [0]ID [1]Tanggal [2]NIS [3]Nama [4]Kelas [5]Kategori [6]Rahasia [7]Isi [8]Guru [9]Jenis Kesalahan
      if (data[i][2] === nis && data[i][5] === kategori) {
        jumlah++;
        var tgl = data[i][1];
        
        // 2. KUNCI PERBAIKAN: Hilangkan spasi tersembunyi dengan .trim()
        var jns = (data[i][9] || "").toString().trim(); 
        var isiCatatan = (data[i][7] || "").toString().trim(); // Mengambil kolom "Isi"
        
        // 3. KUNCI PERBAIKAN: Jadikan huruf kecil semua agar cocok 100%
        if (jns === "" || jns.toLowerCase() === "lainnya") {
          jns = isiCatatan;
          
          if (jns.length > 50) {
            jns = jns.substring(0, 47) + "...";
          }
        }
        
        daftarKejadian.push("▪️ " + tgl + " : " + jns);
      }
    }

    Logger.log("cekDanIngatkanGuruAnekdot: NIS=" + nis + " kategori=" + kategori + " total=" + jumlah);

    if (jumlah > 0 && jumlah % AMBANG_PENGINGAT === 0) {
      var rincianKejadian = daftarKejadian.join("\n");
      kirimPengingatGuruAnekdot(nama, kelas, kategori, jumlah, rincianKejadian);
    }
  } catch (err) {
    Logger.log("ERROR cekDanIngatkanGuruAnekdot: " + err.toString());
  }
}

    }

    UrlFetchApp.fetch("https://api.fonnte.com/send", {

  }
}

function pastikanKolomJenisKesalahan(sheetAnekdot) {
  var lastCol = sheetAnekdot.getLastColumn();
  var headers = lastCol > 0 ? sheetAnekdot.getRange(1, 1, 1, lastCol).getDisplayValues()[0] : [];
  if (headers.indexOf("Jenis Kesalahan") === -1) {
    sheetAnekdot.getRange(1, lastCol + 1).setValue("Jenis Kesalahan");
  }
}

function formatTampilanSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ss.getSheets();
  
  for (var i = 0; i < sheets.length; i++) {
    var sheet = sheets[i];
    var sheetName = sheet.getName().toLowerCase();
    
    terapkanDesainProfesional(sheet);
    if (sheetName.indexOf("absen") !== -1) {
      terapkanWarnaKehadiran(sheet);
    }
  }
}

function terapkanDesainProfesional(sheet) {
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow === 0 || lastCol === 0) return; 
  
  sheet.setFrozenRows(1);
  
  var headerRange = sheet.getRange(1, 1, 1, lastCol);
  headerRange.setBackground("#0f172a") 
             .setFontColor("#ffffff")
             .setFontWeight("bold")
             .setHorizontalAlignment("center")
             .setVerticalAlignment("middle");
             
  if (lastRow > 1) {
    var dataRange = sheet.getRange(2, 1, lastRow - 1, lastCol);
    dataRange.setVerticalAlignment("middle");
    
    var allRange = sheet.getRange(1, 1, lastRow, lastCol);
    allRange.setBorder(true, true, true, true, true, true, "#cbd5e1", SpreadsheetApp.BorderStyle.SOLID);
    
    var bandings = sheet.getBandings();
    for (var b = 0; b < bandings.length; b++) {
      bandings[b].remove();
    }
    dataRange.applyRowBanding(SpreadsheetApp.BandingTheme.LIGHT_GREY, false, false);
  }
  
  for (var col = 1; col <= lastCol; col++) {
    sheet.autoResizeColumn(col);
    var width = sheet.getColumnWidth(col);
    sheet.setColumnWidth(col, width + 15);
  }
}

function terapkanWarnaKehadiran(sheet) {
  sheet.clearConditionalFormatRules();
  
  var maxRow = Math.max(sheet.getMaxRows(), sheet.getLastRow() + 100); 
  var maxCol = sheet.getMaxColumns();
  var range = sheet.getRange(2, 1, maxRow, maxCol); 
  
  function buatRule(teks, bgWarna, fontWarna) {
    return SpreadsheetApp.newConditionalFormatRule()
      .whenTextEqualTo(teks)
      .setBackground(bgWarna)
      .setFontColor(fontWarna)
      .setRanges([range])
      .build();
  }
  
  var rules = [];
  rules.push(buatRule("Hadir", "#d1fae5", "#065f46"));
  rules.push(buatRule("Sakit", "#fef3c7", "#92400e"));
  rules.push(buatRule("Izin", "#fef3c7", "#92400e"));
  rules.push(buatRule("Alpa", "#fee2e2", "#991b1b"));
  
  sheet.setConditionalFormatRules(rules);
}
